# NETWORKS E PGADMIN

# Pgadmin é uma melhor forma de interagir com o banco, um gerenciador de bancos.
# vamos criar uma rede (network) para que os containers (do banco e do gerenciador pgadmin)
# possam se comunicar / se ver

-- Cria a rede pg-network
 docker network create pg-network

-- (Re) Cria o container do banco atrelando a rede e dando um nome (pg-database)
 docker run -it \
    -e POSTGRES_USER="root" \
    -e POSTGRES_PASSWORD="root" \
    -e POSTGRES_DB="ny_taxi" \
    -v C:/Users/elton.junior_amaris/Desktop/ej_de_zoomcamp/ny_taxi_postgres_data:/var/lib/postgresql/data \
    -p 5432:5432 \
    --network=-network \
    --name=pg-database \
    postgres:13

-- Cria o container do pgamin 

docker run -it \
    -e PGADMIN_DEFAULT_EMAIL="admin@admin.com" \
    -e PGADMIN_DEFAULT_PASSWORD="root" \
    -p 8080:80 \
    --network=pg-network \
    --name pg-admin \
    dpage/pgadmin4 

--- ACESSAR O BANCO PELO PGADMIN

-- Com os containers ativos, entrar no browser o endereço do pgadmin (http://localhost:8080/)
-- fazer login com o email e senha e, depois, conectar no container do banco atravé de seu nome no campo host (pg-database)
-- mais as infos de usuario, senha, porta, etc.
